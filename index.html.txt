        const menuBtn = document.getElementById('menu-btn');

        function copyToClipboard(text) {
            navigator.clipboard.writeText(text).then(() => {
                alert('PSN ID "' + text + '" wurde in die Zwischenablage kopiert!');
            }).catch(() => {
                alert('PSN ID: ' + text);

        function startCountdown() {
            let hours = 18, mins = 45, secs = 30;
            setInterval(() => {
                if (secs > 0) {
                    secs--;
                } else {
                    secs = 59;
                    if (mins > 0) {
                        mins--;
                    } else {
                        mins = 59;
                        if (hours > 0) hours--;
                    }
                }
                document.getElementById('cd-hours').innerText = hours < 10 ? '0' + hours : hours;
                document.getElementById('cd-mins').innerText = mins < 10 ? '0' + mins : mins;
                document.getElementById('cd-secs').innerText = secs < 10 ? '0' + secs : secs;
            }, 1000);
        }

        const defaultMessages = [
            { name: "GamerBro_TR", message: "Starker Live Stream gestern! Die Warzone Siege waren heftig! 🔥", date: "Vor 1 Stunde" },
            { name: "PS5_Squad", message: "Habe dich auf PSN geaddet (Antepli27). Lass mal am Wochenende FC 25 zocken!", date: "Vor 4 Stunden" }
        ];

        function loadGuestbook() {
            const stored = localStorage.getItem('antepli27_gb');
            const entries = stored ? JSON.parse(stored) : defaultMessages;
            const container = document.getElementById('gb-entries-list');

            container.innerHTML = '';
            entries.forEach(entry => {
                const card = document.createElement('div');
                card.className = 'glass-card p-4 rounded-2xl border border-neon-cyan/10';
                card.innerHTML = `
                    <div class="flex justify-between items-center mb-1">
                        <span class="font-heading font-bold text-neon-cyan text-sm">${escapeHTML(entry.name)}</span>
                        <span class="text-[10px] text-gray-500">${entry.date}</span>
                    </div>
                    <p class="text-gray-300 text-sm">${escapeHTML(entry.message)}</p>
                `;
                container.appendChild(card);
            });
        }

        function addGuestbookEntry(e) {
            e.preventDefault();
            const name = document.getElementById('gb-name').value;
            const message = document.getElementById('gb-message').value;

            const stored = localStorage.getItem('antepli27_gb');
            const entries = stored ? JSON.parse(stored) : [...defaultMessages];

            entries.unshift({ name, message, date: "Gerade eben" });
            localStorage.setItem('antepli27_gb', JSON.stringify(entries));

            document.getElementById('gb-name').value = '';
            document.getElementById('gb-message').value = '';

            loadGuestbook();
        }

        function escapeHTML(str) {
            return str.replace(/[&<>'"]/g, 
                tag => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', "'": '&#39;', '"': '&quot;' }[tag] || tag)
            );
        }

        window.onload = function() {
            startCountdown();
            loadGuestbook();
        };
    </script>
